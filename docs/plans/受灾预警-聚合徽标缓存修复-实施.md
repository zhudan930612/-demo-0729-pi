```ts
export function warningRenderSignature(entries: WarnedVillageEntry[]): string {
  return entries
    .filter((entry) => entry.level >= 2)
    .map((entry) => `${entry.village.code}:${entry.level}`)
    .sort()
    .join(',')
}
```
